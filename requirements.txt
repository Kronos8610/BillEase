# BillEase — dependencias
# Requiere Python 3.11 o superior: la base cifrada se carga en memoria con
# `sqlite3.Connection.deserialize`, que llegó en la 3.11.
#
#   python -m venv .venv
#   .venv\Scripts\activate        (Windows)
#   source .venv/bin/activate     (macOS / Linux)
#   pip install -r requirements.txt

# --- Aplicación ---
PyQt6==6.9.1
reportlab==4.4.10

# Cifrado de la contraseña del autónomo y derivación de la clave de la caja.
argon2-cffi==25.1.0

# AES-256-GCM para cifrar la base de datos (`core/cifrado.py`). Se usa solo
# `AESGCM`, que no ha cambiado desde hace años. La alternativa habitual,
# SQLCipher, no publica ruedas para Windows en ninguna versión: pedirle a un
# autónomo que compile una biblioteca en C para facturar no es una opción.
cryptography==46.0.3

# --- Desarrollo y pruebas ---
pytest==8.4.2
pytest-qt==4.5.0
pytest-cov==7.0.0

# Lee el PDF generado en las pruebas: sirve para comprobar que ningún
# concepto se dibuja fuera del papel y que los totales impresos cuadran.
pypdf==6.18.1

# --- Empaquetado ---
# Solo hace falta para generar el ejecutable:
#   pyinstaller billease.spec
# El instalador se compila aparte, con Inno Setup 6 (installer/billease.iss).
pyinstaller==6.22.3
